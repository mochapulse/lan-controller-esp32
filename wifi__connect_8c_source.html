53static esp_err_t apply_static_ip(const char *static_ip)
54{
55 if (!static_ip || static_ip[0] == '\0') {
56 return ESP_OK; /* DHCP default */
57 }
58
59 esp_ip4_addr_t ip;
60 if (esp_netif_str_to_ip4(static_ip, &ip) != ESP_OK) {
61 ESP_LOGW(TAG, "Invalid static IP '%s' — falling back to DHCP",
62 static_ip);
63 return ESP_OK;
64 }
65
66 esp_netif_ip_info_t ip_info = { 0 };
67 ip_info.ip = ip;
68
69 char gw_str[16];
70 snprintf(gw_str, sizeof(gw_str), "%d.%d.%d.1",
71 esp_ip4_addr1(&ip), esp_ip4_addr2(&ip), esp_ip4_addr3(&ip));
72 if (esp_netif_str_to_ip4(gw_str, &ip_info.gw) != ESP_OK ||
73 esp_netif_str_to_ip4("255.255.255.0", &ip_info.netmask) != ESP_OK) {
74 ESP_LOGW(TAG, "Failed to build static IP info — falling back to DHCP");
75 return ESP_OK;
76 }
77
78 esp_err_t ret = esp_netif_dhcpc_stop(s_sta_netif);
79 if (ret != ESP_OK) {
80 ESP_LOGE(TAG, "Failed to stop DHCP client: %s", esp_err_to_name(ret));
81 return ret;
82 }
83
84 ret = esp_netif_set_ip_info(s_sta_netif, &ip_info);
85 if (ret != ESP_OK) {
86 ESP_LOGE(TAG, "Failed to set static IP: %s", esp_err_to_name(ret));
87 return ret;
88 }
89
90 ESP_LOGI(TAG, "Static IP configured: %s (gw " IPSTR ", netmask " IPSTR ")",
91 static_ip, IP2STR(&ip_info.gw), IP2STR(&ip_info.netmask));
92 return ESP_OK;
93}

104static void event_handler(void *arg,
105 esp_event_base_t event_base,
106 int32_t event_id,
107 void *event_data)
108{
109 if (event_base == WIFI_EVENT && event_id == WIFI_EVENT_STA_START) {
110 esp_wifi_connect();
111 } else if (event_base == WIFI_EVENT &&
112 event_id == WIFI_EVENT_STA_DISCONNECTED) {
113 xEventGroupClearBits(s_wifi_event_group, WIFI_GOT_IP_BIT);
114 s_retrying = false;
115 s_ip_str[0] = '\0';
116 ESP_LOGI(TAG, "Disconnected from AP");
117 } else if (event_base == IP_EVENT &&
118 event_id == IP_EVENT_STA_GOT_IP) {
119 xEventGroupSetBits(s_wifi_event_group, WIFI_GOT_IP_BIT);
120 s_retrying = false;
121 ip_event_got_ip_t *event = (ip_event_got_ip_t *)event_data;
122 snprintf(s_ip_str, sizeof(s_ip_str),
123 IPSTR, IP2STR(&event->ip_info.ip));
124 ESP_LOGI(TAG, "Got IP: %s", s_ip_str);
125 }
126}

130esp_err_t wifi_init(const char *ssid, const char *password,
131 const char *static_ip)
132{
133 if (!ssid || !password) {
134 return ESP_ERR_INVALID_ARG;
135 }
136
137 /* NVS — erase and re-init if the partition is corrupted or outdated */
138 esp_err_t ret = nvs_flash_init();
139 if (ret == ESP_ERR_NVS_NO_FREE_PAGES ||
140 ret == ESP_ERR_NVS_NEW_VERSION_FOUND) {
141 ESP_ERROR_CHECK(nvs_flash_erase());
142 ret = nvs_flash_init();
143 }
144 if (ret != ESP_OK) {
145 return ret;
146 }
147
148 ESP_ERROR_CHECK(esp_netif_init());
149 ESP_ERROR_CHECK(esp_event_loop_create_default());
150 s_sta_netif = esp_netif_create_default_wifi_sta();
151 if (!s_sta_netif) {
152 ESP_LOGE(TAG, "Failed to create default Wi-Fi station netif");
153 return ESP_FAIL;
154 }
155
156 ret = apply_static_ip(static_ip);
157 if (ret != ESP_OK) {
158 return ret;
159 }
160
161 s_wifi_event_group = xEventGroupCreate();
162 if (!s_wifi_event_group) {
163 ESP_LOGE(TAG, "Failed to create event group");
164 return ESP_ERR_NO_MEM;
165 }
166
167 wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
168 ESP_ERROR_CHECK(esp_wifi_init(&cfg));
169
170 ESP_ERROR_CHECK(esp_event_handler_instance_register(
171 WIFI_EVENT, ESP_EVENT_ANY_ID, &event_handler, NULL, NULL));
172 ESP_ERROR_CHECK(esp_event_handler_instance_register(
173 IP_EVENT, IP_EVENT_STA_GOT_IP, &event_handler, NULL, NULL));

175 wifi_config_t wifi_config = {
176 .sta = {
177 .scan_method = WIFI_FAST_SCAN,
178 .sort_method = WIFI_CONNECT_AP_BY_SIGNAL,
179 .threshold.authmode = WIFI_AUTH_WPA2_PSK,
180 },
181 };
182 strncpy((char *)wifi_config.sta.ssid, ssid,
183 sizeof(wifi_config.sta.ssid) - 1);

187 ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_STA));
188 ESP_ERROR_CHECK(esp_wifi_set_config(WIFI_IF_STA, &wifi_config));

190 ESP_LOGI(TAG, "WiFi init done for SSID: %s", ssid);
191 return ESP_OK;

229esp_err_t wifi_retry(void)
230{
231 if (wifi_health()) {
232 s_retrying = false;
233 return ESP_OK;
234 }
235
236 /* Guard — a connection attempt is already in flight */
237 if (s_retrying) {
238 return ESP_OK;
239 }
240
241 s_retrying = true;
242 ESP_LOGW(TAG, "Retrying WiFi connection…");
243
244 esp_err_t ret = esp_wifi_connect();
245 if (ret == ESP_ERR_WIFI_NOT_STARTED) {
246 /* Station is stopped — start it; the event handler will connect */
247 ret = esp_wifi_start();
248 if (ret != ESP_OK) {
249 ESP_LOGE(TAG, "WiFi start failed: %s", esp_err_to_name(ret));
250 s_retrying = false;
251 return ret;
252 }
253 /* Connection will be driven by WIFI_EVENT_STA_START → handler */
254 return ESP_OK;
255 }
256
257 if (ret != ESP_OK) {
258 ESP_LOGE(TAG, "WiFi connect failed: %s", esp_err_to_name(ret));
259 s_retrying = false;
260 }
261 return ret;
262}
