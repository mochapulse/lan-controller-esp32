104static bool heap_is_safe_or_reject(httpd_req_t *req)

118static bool api_auth_or_reject(httpd_req_t *req)
119{
120 const char *expected_token = dotenv_get("WEB_API_TOKEN");
121 if (!expected_token || expected_token[0] == '\0') {
122 ESP_LOGE(TAG, "WEB_API_TOKEN not configured");
123 httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR,
124 "API auth is not configured");
125 return false;
126 }
127
128 size_t expected_len = strlen(expected_token);
129 if (expected_len == 0 || expected_len >= MAX_API_KEY_LEN) {
130 ESP_LOGE(TAG, "WEB_API_TOKEN length is invalid");
131 httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR,
132 "API auth config is invalid");
133 return false;
134 }
135
136 size_t header_len = httpd_req_get_hdr_value_len(req, API_KEY_HEADER);
137 if (header_len != expected_len) {
138 httpd_resp_send_err(req, HTTPD_401_UNAUTHORIZED, "Unauthorized");
139 return false;
140 }
141
142 char header_val[MAX_API_KEY_LEN] = {0};
143 if (httpd_req_get_hdr_value_str(req, API_KEY_HEADER,
144 header_val, sizeof(header_val)) != ESP_OK) {
145 httpd_resp_send_err(req, HTTPD_401_UNAUTHORIZED, "Unauthorized");
146 return false;
147 }
148
149 if (mbedtls_ct_memcmp(header_val, expected_token, expected_len) != 0) {
150 httpd_resp_send_err(req, HTTPD_401_UNAUTHORIZED, "Unauthorized");
151 return false;
152 }
153
154 return true;
155}

165static esp_err_t static_handler(httpd_req_t *req)
166{
167 for (size_t i = 0; i < sizeof(s_files) / sizeof(s_files[0]); i++) {
168 if (strcmp(req->uri, s_files[i].uri) != 0) continue;
169
170 size_t len = (size_t)(s_files[i].end - s_files[i].start);
171 httpd_resp_set_type(req, s_files[i].content_type);
172 httpd_resp_set_hdr(req, "Cache-Control", "no-cache, no-store, must-revalidate");
173 httpd_resp_send(req, (const char *)s_files[i].start, len);
174 return ESP_OK;
175 }
176
177 httpd_resp_send_404(req);
178 return ESP_FAIL;
179}

281
282 char *json_str = cJSON_PrintUnformatted(root);
283 cJSON_Delete(root);
284
285 httpd_resp_set_type(req, "application/json");
286 httpd_resp_send(req, json_str, strlen(json_str));

288 free(json_str);
