61void dotenv_parse_line(const char *line, size_t len)
62{
63 const char *eq = memchr(line, '=', len);
64 if (!eq) return;
65
66 size_t key_len = eq - line;
67 while (key_len > 0 && line[key_len - 1] == ' ') key_len--;
68
69 const char *val = eq + 1;
70 size_t val_len = len - (size_t)(val - line);
71 while (val_len > 0 && *val == ' ') { val++; val_len--; }
72
73 if (val_len >= 2 && val[0] == '"' && val[val_len - 1] == '"') {
74 val++;
75 val_len -= 2;
76 }
77
78 while (val_len > 0 && val[val_len - 1] == ' ') val_len--;
79 while (val_len > 0 && val[val_len - 1] == '\r') val_len--;
80
81 if (key_len == 0 || key_len >= DOTENV_KEY_LEN) return;
82 if (val_len >= DOTENV_VAL_LEN) return;
83 if (s_entry_count >= DOTENV_MAX_KEYS) return;
84
85 memcpy(s_entries[s_entry_count].key, line, key_len);
86 s_entries[s_entry_count].key[key_len] = '\0';
87 memcpy(s_entries[s_entry_count].val, val, val_len);
88 s_entries[s_entry_count].val[val_len] = '\0';
89 s_entry_count++;
90}

94void dotenv_init(void)
95{
96 const unsigned char *p = _binary__env_start;
97 const unsigned char *end = _binary__env_end;
98
99 ESP_LOGI(TAG, "Loading embedded .env (%d bytes)", (int)(end - p));
100
101 /*
102 * Walk the blob byte-by-byte.
103 * Lines end at '\n' or '\0'; the *last* line is handled after the loop.
104 * Leading whitespace is stripped per-line; '#' lines are skipped.
105 */
106
107 const unsigned char *line_start = p;
108
109 for (; p < end; p++) {
110 if (*p == '\n' || *p == '\0') {
111 const unsigned char *ls = line_start;
112 while (ls < p && (*ls == ' ' || *ls == '\t')) ls++;
113 size_t len = (size_t)(p - ls);
114 if (len > 0 && *ls != '#') {
115 dotenv_parse_line((const char *)ls, len);
116 }
117 line_start = p + 1;
118 if (*p == '\0') break;
119 }
120 }
121
122 /* Handle the final line (no trailing newline / NUL at EOF) */
123 if (line_start < end) {
124 const unsigned char *ls = line_start;
125 while (ls < end && (*ls == ' ' || *ls == '\t')) ls++;
126 size_t len = (size_t)(end - ls);
127 if (len > 0 && *ls != '#') {
128 dotenv_parse_line((const char *)ls, len);
129 }
130 }
131
132 ESP_LOGI(TAG, "Parsed %d entries", s_entry_count);
133}
